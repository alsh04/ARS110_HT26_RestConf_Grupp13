############# HEMMA ################

enable
configure terminal
hostname hemma
no ip domain lookup
ip domain name cisco
service password-encryption
enable secret Network!337
username admin privilege 15 secret Network!337
vrf definition Mgmt-vrf
address-family ipv4
exit-address-family
exit

interface GigabitEthernet0/0/1
description OOB management
vrf forwarding Mgmt-vrf
ip address 192.168.100.11 255.255.255.0
no shutdown
exit

ip ssh version 2
ip http secure-server
ip http authentication local
restconf
line console 0
logging synchronous
login local
exit
line vty 0 15
logging synchronous
login local
transport input ssh
exit
end
crypto key generate rsa modulus 2048
